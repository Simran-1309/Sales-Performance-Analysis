USE SalesAnalysisDB;

-- 1. Profitability by Discount Band
SELECT
    CASE
        WHEN Discount = 0 THEN '0%'
        WHEN Discount <= 0.10 THEN '1-10%'
        WHEN Discount <= 0.20 THEN '11-20%'
        WHEN Discount <= 0.30 THEN '21-30%'
        WHEN Discount <= 0.40 THEN '31-40%'
        WHEN Discount <= 0.50 THEN '41-50%'
        WHEN Discount <= 0.60 THEN '51-60%'
        ELSE 'Above 60%'
    END AS Discount_Band,
    COUNT(*) AS Total_Records,
    ROUND(AVG(Discount) * 100, 2) AS Average_Discount_Percent,
    ROUND(SUM(Sales), 2) AS Total_Sales,
    ROUND(SUM(Profit), 2) AS Total_Profit,
    ROUND(
        SUM(Profit) / NULLIF(SUM(Sales), 0) * 100,
        2
    ) AS Profit_Margin_Percent
FROM SalesData
GROUP BY Discount_Band
ORDER BY MIN(Discount);

-- 2. Order-to-Ship Time by Shipping Mode
SELECT
    Ship_Mode,
    COUNT(DISTINCT Order_ID) AS Total_Orders,
    ROUND(AVG(DATEDIFF(Ship_Date, Order_Date)), 2)
        AS Average_Ship_Days,
    MIN(DATEDIFF(Ship_Date, Order_Date)) AS Minimum_Ship_Days,
    MAX(DATEDIFF(Ship_Date, Order_Date)) AS Maximum_Ship_Days
FROM SalesData
GROUP BY Ship_Mode
ORDER BY Average_Ship_Days;

-- 3. Sales and Profit by Shipping Mode
SELECT
    Ship_Mode,
    COUNT(DISTINCT Order_ID) AS Total_Orders,
    ROUND(SUM(Sales), 2) AS Total_Sales,
    ROUND(SUM(Profit), 2) AS Total_Profit
FROM SalesData
GROUP BY Ship_Mode
ORDER BY Total_Orders DESC;

-- 4. Discount and Profit by Category
SELECT
    Category,
    ROUND(AVG(Discount) * 100, 2) AS Average_Discount_Percent,
    ROUND(SUM(Sales), 2) AS Total_Sales,
    ROUND(SUM(Profit), 2) AS Total_Profit,
    ROUND(
        SUM(Profit) / NULLIF(SUM(Sales), 0) * 100,
        2
    ) AS Profit_Margin_Percent
FROM SalesData
GROUP BY Category
ORDER BY Average_Discount_Percent DESC;
